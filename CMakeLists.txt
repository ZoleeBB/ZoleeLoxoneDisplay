cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET esp32p4)
set(PANEL_SKU "31416" CACHE STRING "Waveshare SKU (31416, 31570)")
set(PANEL_SILICON "rev1" CACHE STRING "ESP32-P4 silicon (rev1, rev3)")
if(NOT PANEL_SKU MATCHES "^(31416|31570)$" OR NOT PANEL_SILICON MATCHES "^(rev1|rev3)$")
    message(FATAL_ERROR "Unsupported PANEL_SKU / PANEL_SILICON")
endif()
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig-${PANEL_SKU}-${PANEL_SILICON}")
set(SDKCONFIG_DEFAULTS "sdkconfig.defaults;profiles/${PANEL_SKU}.defaults;profiles/${PANEL_SILICON}.defaults")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(zolee_loxone_display)
include(cmake/display_psram.cmake)
include(cmake/hosted_rx.cmake)
